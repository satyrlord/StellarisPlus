	}
}

	script = jobs/zone_foundry_add

triggered_planet_modifier = {
	potential = {
		industrial_jobs_are_catalytic_trigger = no
	}
	modifier = {
		planet_metallurgists_minerals_upkeep_add = 1.5
	}
}

# Catalytic Processing empires pay it in food instead, at the same 1.5x ratio the base job uses when swapping minerals for food
triggered_planet_modifier = {
	potential = {
		industrial_jobs_are_catalytic_trigger = yes
	}
	modifier = {
		planet_metallurgists_food_upkeep_add = 2.25
	}
